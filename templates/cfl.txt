# Файл обязателен для Euler при stabilization = cfl, low или on.
# Все пять параметров обязательны: cfl_num_steps, cfl_coeff, cfl_init, cfl_min, cfl_max.
# Значений по умолчанию нет.

# Требуются целое cfl_num_steps >= 1, cfl_coeff > 1 и 0 < cfl_min <= cfl_init <= cfl_max.
# При переходе со стартового первого порядка на выбранную реконструкцию CFL возвращается
# к cfl_init, счётчик принятых шагов для повышения CFL начинается заново. Если границы
# изменили в control.txt, солвер ограничивает cfl_init действующим диапазоном.
# После cfl_num_steps принятых шагов повышаем CFL в cfl_coeff раз до cfl_max.
# При отказе из-за нефизического состояния газа делим CFL на cfl_coeff и повторяем весь шаг.
cfl_num_steps = 10
cfl_coeff = 2
cfl_max = 1
# Начальный CFL и нижний предел. Ровно на пределе повтор ещё разрешён.
cfl_init = 0.5
cfl_min = 0.001
