# Файл обязателен при Turbulence = On. Все параметры безразмерны.
# model обязателен: RANS, LES или DDES. Ниже приведена готовая настройка RANS.
model = RANS
# rans_eq обязателен для RANS и DDES.
rans_eq = SA

# В BDF1, BDF2 и steady внутренние итерации переноса SA заканчиваются при достижении
# допуска или после max(200, newton_max_iter) поправок. Далее используется полученная рабочая вязкость.

# Для RANS SA и DDES SA обязательно отношение физической турбулентной вязкости к молекулярной
# в набегающем потоке. Допускается ноль (ламинарное решение). Рабочую величину
# nu_tilde солвер вычисляет из этого отношения с учётом демпфирования fv1.
nut_nu_amb = 0.21

# Для LES замените строки model, rans_eq и nut_nu_amb на следующие три строки:
# model = LES
# les_eq = Sigma
# subgridscale = sla
# les_eq обязателен для LES. subgridscale обязателен для LES и DDES. Все шесть вариантов применяются к исходному элементу:
# max, min — длина самого длинного или короткого ребра; vol — кубический корень из объёма.
# Delta_omega — квадратный корень из среднего сечения поперёк завихренности (объём / толщина вдоль оси).
# Delta_omega_tilde (также Delta_omega^tilde) — диаметр проекции вершин поперёк вихря, делённый на √3.
# sla — Delta_omega_tilde с множителем F_KH. Если ротор нулевой или неотличим от округления, базовый масштаб берётся из объёма.
# switch_on_vtm = on   # Необязателен, по умолчанию on. В sla значение off даёт ровно Delta_omega_tilde.
# F_KH_lim = 0.1       # Необязателен, от 0 до 1. Нижняя граница F_KH, по умолчанию 0.1.
# VTM кинематический, усредняется по элементам узла. Между VTM=0.15 и 0.3 множитель растёт линейно до 1.
# c_sigma = 1.35       # Необязательная положительная константа Sigma, по умолчанию 1.35.
# В 1D и 2D модель даёт nut=0. В 3D значения nut рассчитываются из градиента скорости.

# Для SA-DDES в примере RANS замените model = RANS на model = DDES и добавьте
# subgridscale = max    # Любой из шести перечисленных выше масштабов.
# c_des = 0.65          # Необязательный положительный коэффициент, по умолчанию 0.65.
# На узле берётся среднеквадратический масштаб соседних элементов с весом по объёму.
# Для sla общий множитель F_KH применяется к этому масштабу. При F_KH=0 и отсутствии
# стенок предельное значение рабочей вязкости равно нулю. В 2D объёмный масштаб
# равен корню из площади, в 1D — длине. Он также заменяет Δω при нулевой толщине вдоль вихря.
# Для защиты пограничного слоя используется норма полного градиента скорости:
# её среднее по соседним элементам берётся с теми же объёмными весами.
# У стены с прилипанием рабочая вязкость равна нулю. Вдали от стенки
# модель использует C_DES * Δ. Поля nu_tilde и nut доступны как в RANS SA.

# Турбулентное число Прандтля для теплового потока во всех трёх моделях. Необязательно, по умолчанию 0.9.
pr_t = 0.9
