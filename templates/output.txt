# Файл необязателен. Без него запись зон по расписанию отключена.
# Для каждой заданной зоны обязательны имя секции, file и variables, значений по умолчанию нет.
# Обязателен ровно один интервал: every_steps или every_time. Интервала по умолчанию нет.
# Необязательные параметры зоны: multi = 0, format = binary, layout = local по умолчанию.

# Каждая секция задаёт вывод полей по своему расписанию. Для Cp выбирается одна метка стенки.
# При завершении и по STOP все зоны записываются на последнем принятом шаге.
# Для multi=1 этот номер входит в имя файла, в том числе при завершении на шаге 0.
# Пустой или отсутствующий output.txt отключает плановую запись полей.
# file — относительный путь внутри каталога вывода выбранного layout.
# Выберите ровно один интервал: every_steps (целое > 0, принятые шаги)
# или every_time (число > 0, расчётное время).
# variables — список величин и групп через запятую без повторов. Основные величины: rho, ux, uy, uz, p, E.
# При Turbulence = On доступна nut (турбулентная кинематическая вязкость).
# Для SA доступна также nu_tilde (рабочая вязкость). Обе величины выражены в единицах U_ref*L_ref.
# uy доступна с 2D, uz — в 3D; E — полная энергия на единицу объёма.
# multi — 0 (по умолчанию, заменять файл) или 1 (номер шага в имени).
# format — binary (по умолчанию), text или lzma.
# layout — local (по умолчанию, отдельные файлы рангов) или global
# (общий файл в исходной нумерации). Для общего вывода MPI используйте
# layout = global, format = binary и сетку .mdec версии 4.

[latest]
file = latest.fields
every_steps = 100
multi = 0
variables = rho, ux, uy, p, E
# format = binary
# layout = local

[pressure_history]
file = pressure.fields
every_time = 0.05
multi = 1
variables = p
# Например, запись на шаге 37 получит имя pressure_000037.fields.

# В variables можно указывать отдельные величины или группы через запятую.
# primary — rho, компоненты скорости в размерности сетки, p, E.
# fluctuations — соответствующие rho_prime, ux_prime и т.д. В каждом узле из
# мгновенного значения вычитается среднее по времени от начала накопления.
# Интеграл берётся трапециями по всем принятым шагам, независимо от частоты записи.
# В steady это среднее по псевдовремени. При перезапуске история берётся из recovery.
# Если в старом recovery истории нет, накопление начинается с момента продолжения.
# dp_dt (также dp/dt) — мгновенная производная давления по времени в неподвижном узле.
# Для Euler она вычисляется из правых частей плотности, импульса и полной энергии.
# Для Linear берётся правая часть уравнения пульсации давления. На начальном слое
# dp_dt определяется начальным полем и может быть ненулевой. Начальные пульсации
# относительно среднего по времени равны нулю.
# vorticity — модуль ротора скорости. turbulence — nut для LES и пара nu_tilde, nut для SA. Требуется Turbulence=On.
# Например: variables = primary, fluctuations, dp_dt, vorticity
#
# Cp выдаётся в отдельном файле. wall_label обязателен только для variables = Cp.
# В файле только узлы выбранной стенки, их исходные номера с нуля, x, y, z и Cp.
# Cp=(p-p_inf)/q_inf. Для Euler требуется gas.txt: mach, при rho_inf=U_inf=1
# p_inf=1/(gamma*mach^2), q_inf=1/2. Для Linear используется заданный фон.
# [wall_pressure]
# file = wall.fields
# variables = Cp
# wall_label = ymin
# format = binary
# layout = global
# every_steps = 10
