# Заголовок [имя] должен совпадать с меткой границы в сетке.
# Опишите каждую метку ровно один раз. Периодическая секция покрывает две метки.
# bc_type — wall, free или periodic.

# Непроницаемая стенка. Достаточно указать тип.
[ymin]
bc_type wall

# На открытой границе задайте все три строки. Солвер выбирает режим отдельно
# на каждой площадке по текущей скорости вдоль наружной нормали u_n и скорости звука c.
[ymax]
bc_type free
superin initial  # u_n <= -c: взять полное начальное состояние площадки
inflow  base_p   # -c < u_n < 0: задать давление на дозвуковом входе
outflow base_u   # 0 <= u_n < c: задать нормальную скорость на дозвуковом выходе
# При u_n >= c поток целиком определяется внутренним состоянием.
#
# superin без аргументов берёт текущее состояние. Можно заменить отдельные
# примитивные величины, например «superin initial rho=1.2 p=101325 uy=-500».
# Для inflow и outflow доступны base_p (давление), base_rho (плотность)
# и base_u (скорость вдоль наружной нормали). Без аргумента используется
# текущее значение, initial берёт начальное, число задаёт постоянное.

# Периодика. pair — парная метка, translation — три компоненты вектора
# переноса от исходной стороны к парной. Отдельная секция [xmax] не нужна.
[xmin]
bc_type periodic
pair xmax
translation 1 0 0
