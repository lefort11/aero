# Заголовок [имя] должен совпадать с меткой границы в сетке.
# Опишите каждую метку ровно один раз. Периодическая секция покрывает две метки.
# bc_type — wall, free, superin, superout или periodic.

# Непроницаемая стенка. Достаточно указать тип.
[ymin]
bc_type wall

# Открытая граница. Для free обязательны оба правила: inflow и outflow.
# Для каждого доступны base_p (давление), base_rho (плотность), base_u
# (нормальная скорость со знаком). Без аргумента берётся внутреннее
# состояние; initial берёт начальное значение; число задаёт константу.
# Например: inflow base_p initial или outflow base_u 0.2.
# Постоянные давление и плотность должны быть положительными.
[ymax]
bc_type free
inflow  base_p
outflow base_u

# Для сверхзвукового входа доступно inflow superin, для выхода — outflow superout.
# На целой границе можно задать bc_type superin или bc_type superout.
# superout не требует аргументов. Для superin можно указать initial и/или
# заменить отдельные величины rho, p, ux, uy, uz (по размерности сетки).
# Пример: bc_type superin initial rho=1.2 p=101325 ux=500

# Периодика. pair — парная метка, translation — три компоненты вектора
# переноса от исходной стороны к парной. Отдельная секция [xmax] не нужна.
[xmin]
bc_type periodic
pair xmax
translation 1 0 0
