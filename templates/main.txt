# Обязательны без значения по умолчанию: mesh, reconstruction, reconstruction_variables,
# time_scheme, euler_solver и backend.
# cfl обязателен при stabilization = off, значения по умолчанию нет. При cfl, low и on
# стартовое число Куранта задаёт cfl_init из cfl.txt.
# final_time обязателен для euler, ssprk3, BDF1 и BDF2, значения по умолчанию нет.
# При steady его можно опустить: останов определяется критериями из implicit.txt.
#
# Необязательные параметры и их значения по умолчанию:
# Equations = Euler; stabilization = off (для Linear допустим только off); exact_solution = off.
# upw_coeff = 1; t_fo = 0; trace_ratio_min = 0; trace_ratio_max = 1e16.
# omp_threads = 0 (число потоков выбирает среда OpenMP); report_every = 0 (отчёт выключен).
# control_every = 16; recovery_every_steps = 0 (периодическая запись выключена).
# Для MPI: mpi_timing = 0; mpi_rank_weights и mpi_gpu_memory_mib по умолчанию автоматические.
# mpi_node_memory_mib необязателен: при пропуске нет ручного лимита; явное auto включает учёт памяти.
# mpi_memory_factor необязателен: 4 для first_order, 12 для ebr3, 16 для ebr5 и weno3,
# 24 для weno5, mp5 и MPWENO5.

# Готовый пример для 2D на CPU. Из корня проекта: bin/aero --input=templates --output=results
# При копировании шаблонов в каталог задачи укажите путь к своей сетке и её метки в bc.txt.
# Чтобы включить параметр из комментария, уберите начальный #.

# Euler — полные уравнения, Linear — пульсации вокруг однородного фона из linear.txt.
Equations = Euler

# viscosity допускает off (без вязкости), normal или sutherland. Значение по умолчанию — off.
# Для обоих законов задайте в gas.txt числа Рейнольдса и Прандтля и опорную температуру.
# Для обоих законов требуется mach, для sutherland также постоянная Сазерленда.
# Вязкость доступна для Equations = Euler при сборке с каталогом source/viscosity.
# В BDF1 и BDF2 вязкие члены AES входят в невязку и в матрицу поправки Ньютона.
# viscosity = off

# Turbulence — On или Off, по умолчанию Off. Для On нужны Euler, активный закон
# молекулярной вязкости и файл turbulence.txt с параметрами модели RANS SA, LES Sigma или DDES SA.
# Turbulence = Off

# Сетка и численная схема (обязательные параметры)
# mesh — путь относительно этого каталога. Один процесс читает .mesh или .mdec.
# Для нескольких процессов подготовьте .mdec утилитой meshdec.
mesh           = "../qa/cases/momentum_2d/grid.mesh"
# reconstruction — first_order, ebr3, ebr5, weno3, weno5, mp5 или MPWENO5. Для MPWENO5 см. scheme.txt.
# У MP5 и MPWENO5 alpha = min(4, max(0, 1/cfl - 1)) для Эйлера и SSPRK3.
# При BDF1, BDF2 и steady обе схемы используют alpha = 4 при любом CFL.
# Если дальняя точка недоступна, используется WENO3; без ближней точки — среднее на ребре.
reconstruction = ebr3
# reconstruction_variables — выбор переменных реконструкции:
# characteristic (характеристические переменные) или flux (потоки).
# characteristic_flux восстанавливает расщеплённые потоки нелинейного Euler
# в общем базисе ребра для ebr5, weno3, weno5, mp5 и MPWENO5, с Roe или LF.
reconstruction_variables = characteristic
# Энтропийная поправка Роу для двух акустических волн, включая неявный якобиан Роу.
# off (по умолчанию), on (ширина 0.1*c) или коэффициент ширины от 0 до 1.
# При |lambda| < delta используется (lambda²/delta + delta)/2. На LF флаг не влияет.
entropy_fix = off
# time_scheme — euler, ssprk3, BDF1 (неявный Эйлер), BDF2 или steady (установление). Для неявного расчёта заполните implicit.txt.
time_scheme    = ssprk3
# euler_solver — roe или lf (локальный поток Лакса — Фридрихса).
euler_solver   = roe
# cfl — конечное число > 0. final_time — конечное расчётное время >= 0. В режиме steady этот параметр можно опустить.
cfl            = 0.5
# cfl_method необязателен, по умолчанию height. Для чистой конвекции берётся
# минимум по ячейкам CFL*h/max(|u|+c), максимум скорости — по вершинам этой же ячейки.
# h — длина отрезка в 1D, кратчайшая высота треугольника или тетраэдра.
# Для других ячеек h — наименьшая ширина вдоль нормалей к их граням.
# spectral возвращает прежний шаг по сумме скоростей на площадках контрольного объёма.
# В обоих вариантах учитываются ограничения вязкости и переноса SA, если они включены.
cfl_method     = height
final_time     = 2.0
# До t_fo все узлы используют первый порядок по пространству; затем включается reconstruction.
# 0 — выбранная схема с начала расчёта. В steady используется псевдовремя.
t_fo           = 0
# steps_fo необязателен, по умолчанию 0. Первые N принятых шагов (в steady — псевдошагов)
# выполняются с первым порядком. После recovery отсчёт продолжается. Если задан и t_fo,
# выбранная реконструкция включается после достижения обоих порогов.
# В steady на этом же принятом поле солвер заново вычисляет потоки и пространственную
# невязку по reconstruction. Уже эта норма попадает в отчёт и в проверку steady_tol.
# При активной стабилизации первый шаг выбранной реконструкции начинает с cfl_init.
# Если через control.txt изменены границы CFL, cfl_init ограничивается текущим диапазоном.
steps_fo       = 0
# backend = cpu для процессора, gpu для ускорителя текущей сборки.
# Для gpu и hip выберите устройства в gpu.txt, для cuda — в cuda.txt.
backend        = cpu

# Ограничитель реконструкции (показаны значения по умолчанию).
# Допустимые отношения реконструированных плотности и давления к узловым.
# 0 <= trace_ratio_min < 1; trace_ratio_max > 1. Оба числа конечные.
# trace_ratio_min = 0
# trace_ratio_max = 1e16

# Потоки, отчёт и контрольные точки. Интервалы задаются в принятых шагах.
# omp_threads — целое >= 0; при 0 или пропуске используются настройки OpenMP.
omp_threads = 8
# Обязательные параметры выбранного режима проверяются до интегрирования.
# Неизвестный ключ в прочитанном файле выводится с WARNING при старте и в каждом отчёте.
# report_every — целое >= 0; 0 или пропуск отключает отчёт по течению.
report_every = 100

# Необязательный интервал прогноза времени завершения в принятых шагах.
# 0 (по умолчанию) отключает прогноз. Работает независимо от report_every.
# Для Euler, SSPRK3 и BDF прогноз строится по продвижению физического времени
# за последний интервал. Для steady невязка записывается каждый принятый шаг.
# Срок достижения steady_tol оценивается при устойчивом снижении невязки на
# нескольких участках истории. При колебаниях, остановке снижения или слишком
# далёком допуске выводится trend unavailable. После recovery, стартового первого
# порядка и ручного изменения режима история набирается заново. Срок исчерпания
# steady_max_steps оценивается отдельно по реальному времени принятых шагов.
prophecy_every_step = 0
# Солвер создаёт control.txt в каталоге --output и очищает его после чтения команд.
# Примеры команд приведены в templates/control.txt.
# По одной команде на строку: cfl, cfl_min, cfl_max, cfl_coeff, cfl_num_steps, upw_coeff, mp6_hybrid.
# Числовое значение задаётся через =. reset_fail и STOP пишутся отдельными строками без значения.
# В steady также доступны local_time_step и local_time_step_max_ratio.
# Проверка после первого шага запуска и далее каждые control_every шагов; 0 отключает управление.
control_every = 16
# mpi_timing — 0 (по умолчанию) или 1. Включает замеры на CPU и GPU:
# потоки, стадии, обмен halo и потоками, CFL, копирование для вывода и запись.
# mpi_timing = 1
# recovery_every_steps — целое >= 0; 0 или пропуск отключает промежуточные точки.
# При завершении расчёта и по команде STOP решатель всегда сохраняет recovery.
# Файлы recovery-0.bin и recovery-1.bin чередуются. При запуске --recover
# выбирает последнюю исправную запись; --recover 0 и --recover 1 задают слот.
# recovery_every_steps = 100

# Разбиение .mdec при запуске MPI (необязательные параметры)
# mpi_rank_weights — auto, одно положительное число для всех процессов
# или список весов по глобальным рангам, например 3,1. Больший вес даёт
# большую долю работы в пределах доступной памяти и целых листьев.
# На GPU auto и пропуск включают калибровку; на CPU веса по умолчанию равны.
# mpi_rank_weights = auto
# mpi_gpu_memory_mib — auto (по умолчанию), положительный лимит в МиБ
# для всех карт или список по глобальным рангам, например 8192,4096.
# Автоматический бюджет — 95% свободной видеопамяти; лимит может его снизить.
# mpi_gpu_memory_mib = auto
# mpi_node_memory_mib — auto (доступная RAM), положительный лимит в МиБ
# на каждый физический узел или список по узлам в порядке их младшего ранга.
# На CPU явное auto включает учёт доступной памяти при планировании.
# mpi_node_memory_mib = auto
# mpi_memory_factor — положительный множитель оценки оперативной памяти.
# По умолчанию: first_order — 4, ebr3 — 12, ebr5 и weno3 — 16, weno5, mp5 и MPWENO5 — 24.
# mpi_memory_factor = 12

# Доля противопоточной поправки LF или Roe (от 0 до 1).
# Для EBR3 и EBR5 этим же коэффициентом задаётся доля EBR в реконструкции:
# 1 — исходная реконструкция EBR, 0 — центральная часть до применения ограничителя.
upw_coeff = 1

# Необязательный параметр, по умолчанию off. Включает вывод L_2 и L_inf на отчётных шагах и в конце.
# formula — формулы из exact.txt (также допускается прежнее on), fields — снимок по пути из exact.txt.
exact_solution = off

# Стабилизация Euler: off, cfl (откат), low (временный первый порядок), on (постоянный).
stabilization = off
