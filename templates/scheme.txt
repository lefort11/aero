# Файл необязателен. mpweno_ad_corr применяется в MPWENO5, по умолчанию on.
# mp6_hybrid применяется в MP5 и MPWENO5, по умолчанию 0.
# Все параметры этого файла необязательны. Порядок задаётся ключом reconstruction в main.txt.
# В steady режим local_time_step_residual_limit = stencil использует узлы фактически
# построенных строк, в том числе укороченных. Настройки этого ограничителя находятся в implicit.txt.

# Построение шаблона для EBR3/5, WENO3/5, MP5 и MPWENO5.
# straight (по умолчанию) — продолжать ребро по прямой.
# ijk — идти по ряду узлов на четырёхугольниках в 2D и гексаэдрах в 3D.
# pl — продолжать ребро вдоль пристенного слоя четырёхугольников или призм.
# Слои отсчитываются от границ adiabatic_wall и novisc_wall.
# auto — сначала пробовать ijk, затем pl. Если не нашлись обе ближние
# точки, используется straight. При нехватке дальних точек порядок снижается.
stencil_geometry = straight
# Предельное отклонение направления от продолжения ребра при выборе точки.
# Для ijk, pl и auto. Значение в градусах, строго между 0 и 90. По умолчанию 36.
stencil_max_angle = 36

# Антидиссипативная поправка chi_boost в MPWENO5. Допустимы on и off.
mpweno_ad_corr = on
# Вес ограниченной центральной поправки MP5 и MPWENO5: 0 — исходная схема, 1 — полная поправка.
# Для газа и скаляров ограничитель MP5 и MPWENO5 использует кандидат своей схемы.
# Масштаб S равен максимуму модулей четырёх наклонов стороны. При S=0 приращение равно нулю.
# Для p=(v/S)((v-b)/S) при p <= 0 принимается кандидат, при p >= 1e-10 применяется полная MP-проекция.
# Между ними вес кандидата равен (1-t)^2*(1+2*t), где t=p/1e-10.
# При включении гибрида каждая сторона включает в свой интервал принятый базовый след.
mp6_hybrid = 0
