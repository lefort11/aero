# Файл обязателен для BDF1, BDF2 и steady. Для этих режимов обязательны solver,
# jacobian_flux, preconditioner, solver_tol, newton_tol и newton_max_iter.
# Значений по умолчанию у них нет, даже если реализован единственный вариант солвера.
# simplified_jacobian необязателен, по умолчанию off; jacobian_update_every необязателен, по умолчанию 0.
# Только для steady используются необязательные steady_criterion (по умолчанию relative),
# steady_tol (1e-6) и steady_max_steps (10000).

# Параметры неявного шага по времени. Солвер читает файл при time_scheme = BDF1, BDF2 или steady.
# Для MPI задайте сетку .mdec в main.txt. При расчёте на GPU выбирается одна карта на каждый процесс.
solver = BiCGSTAB
# Для матрицы используем реконструкцию первого порядка. Для самой невязки действуют
# reconstruction и euler_solver из main.txt, в том числе при вычислении критерия Ньютона.
# Поток якобиана первого порядка, LF или Roe.
jacobian_flux = LF
# on — фиксировать максимальную скорость LF или матрицу диссипации Роу.
# Для Роу фиксируются собственные значения и векторы среднего состояния.
# Так для LF применяется идея фиксированной диссипативной матрицы из препринта
# П. А. Бахвалова 2020 г., № 79, с. 34–35 (https://keldysh.ru/papers/2020/prep2020_79_eng.pdf).
# off — дифференцировать полный конвективный поток, включая изменение диссипации.
# При обоих значениях вязкая часть якобиана учитывает зависимость вязкости и теплопроводности от температуры.
simplified_jacobian = off
# compact_jacobian необязателен, по умолчанию off. Применяется к вязкой матрице BDF1, BDF2 и steady.
# on оставляет в матрице пары узлов, соединённых ребром сетки. Вклад в диагональ
# для каждой оставленной пары определяется из сохранения потока. off учитывает все пары AES.
# Невязка газа и перенос SA в обоих режимах вычисляются полностью. На вязких сетках
# приближённая матрица может изменить число поправок Ньютона и итераций BiCGSTAB.
# Для расчёта на одной видеокарте солвер сначала проверяет размер массивов.
# При недостатке памяти он может собрать тот же компактный якобиан по координатам
# вершин тетраэдров, призм, пирамид и гексаэдров, в том числе на смешанной сетке.
# Потоки и невязка занимают рабочую память до решения СЛАУ, затем её использует BiCGSTAB.
# Ограничение объёма массивов карты в МиБ задаётся через main.txt:mpi_gpu_memory_mib.
compact_jacobian = off
# BJ — блочный Якоби. CSGS — прямой и обратный проходы блочного Гаусса–Зейделя.
# Процессы MPI обмениваются результатом прямого прохода перед обратным.
preconditioner = BJ
# Относительная точность линейной системы: ||b - A*delta||_inf / ||b||_inf.
solver_tol = 1e-8
# solver_max_iter — положительное целое, по умолчанию 200. Предел итераций BiCGSTAB
# для одной поправки Ньютона в BDF1, BDF2 и steady.
# solver_max_iter = 200
# Относительная точность нелинейного шага: ||F(U)||_inf / ||F(U_n)||_inf.
# Норму берём по всем узлам и компонентам. При нулевой исходной невязке шаг уже решён.
# После достижения допуска или предельного числа поправок шаг принимается с текущим полем.
# В отчёте указывается достигнутая относительная невязка, в том числе при остановке по лимиту.
newton_tol = 1e-6
# Максимальное число поправок Ньютона за один шаг по времени.
newton_max_iter = 20
# off — перед поправками Ньютона проверить начальное приближение предиктора.
# При newton_max_iter = 1 после предиктора доступна одна полноценная поправка.
# on — начинать с принятого газового поля U_n, включая steady с SA и LTS auto.
# Проверка нелинейной невязки и решение уравнения SA действуют в обоих режимах.
newton_without_predictor = off
# simplified_newton — необязательный переключатель для BDF1, BDF2 и steady.
# Раскомментируйте следующую строку, чтобы включить упрощённый метод Ньютона.
# simplified_newton = on
# on — собрать матрицу и предобусловливатель при первой поправке и использовать
# их до конца шага. После каждой поправки пересчитываются потоки, правая часть и её норма.
# off — обновлять матрицу на каждой поправке нелинейной задачи, включая случай N > 0.
# Без переключателя режим определяет jacobian_update_every: 0 — off (по умолчанию), N > 0 — on.
# При явном on значения 0 и 1 дают одну сборку за шаг, N > 1 разрешает хранить матрицу N шагов.
# Частота обновления пространственного якобиана при выборе режима по интервалу. 0 — каждая поправка Ньютона.
# 1 — один раз за шаг, N — один раз за N принятых шагов, если потребовалась поправка.
# При изменении dt временной вклад в диагональ пересчитывается на текущем шаге.
jacobian_update_every = 0

# Максимальная степень экстраполяции начального приближения по принятым слоям.
# 0 отключает историю, 1–6 используют доступные рабочие векторы BiCGSTAB.
# Для BDF1, BDF2 и steady с общим шагом газа, без SA и стабилизации.
# В steady RANS с SA и local_time_step = auto действует отдельный предиктор
# по пространственным балансам. Он автоматически учитывает шаг каждого узла.
# Невязка каждой пробы проверяется по полному уравнению выбранной схемы.
predictor_history_order = 0

# Установление стационарного решения (main.txt: time_scheme = steady).
# В каждом узле делим сумму исходящих потоков на объём. M — максимум модуля
# полученной величины по всем уравнениям и собственным узлам всех процессов.
# relative: M/M_начальное <= steady_tol; absolute: M <= steady_tol.
# local_time_step — необязательный, по умолчанию off. Значения on и auto разрешены только в steady.
# При on каждый узел получает свой псевдошаг CFL с учётом вязкости и ограничения SA.
# auto начинает с такого же шага. После каждого принятого шага учитывает изменения
# плотности, давления, скорости и рабочей вязкости SA. Каждый узел ограничивает свой
# шаг по текущему CFL: height использует высоту ячеек, spectral — скорости на площадках
# контрольного объёма. Оба учитывают вязкость и перенос SA. Отказ BiCGSTAB уменьшает
# все шаги вдвое с повтором от того же поля. После 12 неудачных уменьшений выводится ошибка.
# Для steady с local_time_step = auto следующие цели задаются в долях за принятый шаг.
# Для плотности и давления делитель равен прежнему значению, для скорости — скорости звука.
# Для SA делитель равен сумме молекулярной вязкости и модуля прежней рабочей вязкости.
local_time_step_density_change = 0.01
local_time_step_pressure_change = 0.01
local_time_step_velocity_change = 0.01
local_time_step_sa_change = 0.05
# Узел выбирает максимальное отношение изменения к цели. Множитель шага равен
# обратному квадратному корню из этого отношения в пределах от min_factor до max_factor.
# Допустимы 0 < min_factor <= 1 <= max_factor. После повтора или тяжёлой СЛАУ рост удерживается.
local_time_step_min_factor = 0.5
local_time_step_max_factor = 1.25

# local_time_step_residual_limit применяется при steady и local_time_step = auto.
# Допустимы off (по умолчанию), on, stencil, work и cfl. Режим выбирается при запуске.
local_time_step_residual_limit = off
# off оставляет обычную адаптацию auto по изменению полей. В режиме work к ней
# добавляется общий предел отношения шагов по сходимости и работе BiCGSTAB.
# Для work достаточно общей нормы невязки и числа итераций линейного солвера.
# В режиме cfl адаптация и все ограничения work выполняются при CFL=1.
# После согласования с соседями каждый узловой шаг умножается на текущий CFL.
# При K=1 вся сетка получает общий шаг CFL * dt_min(CFL=1).
# Для on и stencil у каждого узла своя история после принятых шагов.
# Раз в 32 шага проверяются отдельно газ и SA. Рост фиксируется, если невязка
# превышает 1% максимума по всей сетке для тех же уравнений и 105% своего
# предыдущего экспоненциального среднего (вес нового слоя 1/32).
# on ограничивает шаг при росте в самом узле. Для stencil достаточно роста хотя бы
# в одном узле шаблона: берутся концы прилегающих рёбер и узлы их реконструкции.
# Для SA учитываются порядок переноса и наветренная сторона. У границы используется
# доступный шаблон. При проверке процессы MPI обмениваются признаками роста.
# Срабатывание задаёт для следующего шага потолок 75%
# от предыдущего шага с поправкой на CFL, но не ниже предыдущего минимума по сетке.
# Доля местного CFL сохраняется до следующей проверки и восстанавливается в 1,25 раза
# при отсутствии роста. При смене схемы или recovery история узлов набирается заново.
# В режимах work, cfl и stencil действует один и тот же предел K для отношения самого большого
# узлового шага к самому маленькому. Первые 32 принятых шага выбранной реконструкции
# выполняются с K=1. Затем раз в 32 шага солвер сравнивает геометрические средние
# нормы пространственной невязки по всей сетке за первые и последние 16 шагов блока.
# При уменьшении нормы K растёт. Увеличение числа итераций BiCGSTAB по сравнению
# с предыдущим блоком замедляет этот рост. В каждом блоке стоимость равна 32 плюс
# сумма числа итераций BiCGSTAB по всем поправкам Ньютона. Рост K ограничен
# отношением начальной нормы этой реконструкции к последней. При росте или застое
# нормы новый K равен sqrt(K). Предел действует с первого шага следующего блока.
# Переход с первого порядка начинает новое наблюдение на том же поле. В recovery
# сохраняются K, обе нормы, накопители текущего блока и стоимость предыдущего.
# Старый recovery с одной опорной нормой начинает новый блок с K=1 и этой нормой.
# local_time_step_max_ratio дополнительно ограничивает K. Изменение только этого
# параметра через control.txt сохраняет накопленную историю.
# В report строка LTS step ratio limit показывает применённый предел.
# Предел итераций задаётся newton_max_iter, допуск установления — steady_tol.
# Для on и stencil строка LTS residual limit показывает число узлов, где потолок уменьшил шаг
# до применения общего ограничения и согласования с соседями. nodes with stored cap —
# число узлов с сохранённым потолком ниже полного CFL. В MPI оба числа суммируются по владельцам.
# После отказа BiCGSTAB возможны nodes = 0 и nodes with stored cap > 0: уменьшенный шаг
# уже укладывается в сохранённые потолки. Повтор попытки не сбрасывает историю.
# local_time_step_probe_node необязателен, по умолчанию наблюдение выключено.
# Для steady + auto выводит в report фактический dt, max_k |R_k|/V и |d(nu_tilde)/dt|
# и долю разрешённого шага CFL одного узла. Даже при stencil здесь выводятся невязки самого узла. Задайте его глобальный номер из сетки (нумерация с нуля).
# Если отчёт совпал с проверкой stencil, строка LTS probe stencil указывает
# узел шаблона, где обнаружен рост узловой невязки газа или SA.
# Эта оценка участвует в выборе следующего шага.
# local_time_step_probe_node = 0
# В control.txt допустимы local_time_step = off, on, auto. При включении auto
# или чтении recovery начальные узловые шаги снова вычисляются из CFL.
# В steady оба ключа (local_time_step и local_time_step_max_ratio) можно менять в control.txt.
# Команда действует со следующего шага. Ручной выбор сохраняется в recovery.
# Для истории и расписаний (every_time, t_fo) часы отсчитывают сумму минимальных шагов.
# При попадании в заданный момент все шаги уменьшаются пропорционально. Для вывода по
# итерациям используйте every_steps. В report dt_min и dt_max — фактически принятый диапазон.
local_time_step = off
# local_time_step_max_ratio — необязательный, по умолчанию off. Действует на каждом шаге в режимах on и auto.
# Узел получает min(dt_i, local_time_step_max_ratio * dt_min). Минимум общий для всех процессов MPI.
# Допустимы числа от 1 (один шаг на всю сетку) и off (без ограничения разброса).
local_time_step_max_ratio = off
# local_time_step_neighbor_ratio — необязательный предел для steady с local_time_step = on или auto.
# По умолчанию off. Если указать 2, то на каждом ребре больший шаг может превышать меньший
# не более чем вдвое. Уменьшение передаётся дальше по рёбрам, пока условие не выполнится везде.
# Число 1 задаёт один общий шаг. Обычный предел local_time_step_max_ratio действует независимо.
# В report строка LTS edge ratio показывает максимальное отношение между соседями до и после.
local_time_step_neighbor_ratio = off
# При relative знаменатель — норма невязки исходного поля на шаге 0, в том числе после --recover.
steady_criterion = relative
# steady_multigrid — необязательный, по умолчанию off. Значение on допустимо только для steady.
# При подготовке солвер объединяет соседние контрольные объёмы в агрегаты.
# Правая часть грубой системы — сумма невязок выбранной схемы. Поправку проверяет
# полная пространственная невязка газа и, при включённом SA, его уравнения переноса.
# Грубый оператор PᵀJP использует пространственную часть уже собранного якобиана.
# Сначала солвер проверяет плавное направление по последнему изменению газа (last step).
# Его множитель подбирается по балансам агрегатов и может достигать 2. Затем при необходимости
# решает блочную грубую систему и сглаживает поправку выбранным BJ или CSGS (aggregates).
# В логе указаны направление, принятый множитель и норма до и после пробы. При отказе
# сохраняется поле обычного псевдошага. Для on нужен каталог source/multigrid при сборке.
# Построение агрегатов, операторы переноса и критерий приёма описаны в source/multigrid/README.md.
# Для SA на подробной сетке после неудачной прогонки по линиям включаются итерации
# Чебышёва. Допуск переноса прежний, невязка проверяется после каждой поправки.
steady_multigrid = off
steady_tol = 1e-6
# Общее число псевдошагов с начала расчёта, включая шаги до перезапуска. При исчерпании записываются поля и возвращается ошибка.
steady_max_steps = 10000
