# Параметры неявного шага по времени. Солвер читает файл при time_scheme = BDF1, BDF2 или steady.
# Для MPI задайте сетку .mdec в main.txt. При расчёте на GPU выбирается одна карта на каждый процесс.
solver = BiCGSTAB
# Для матрицы используем реконструкцию первого порядка. Для самой невязки действуют
# reconstruction и euler_solver из main.txt, в том числе при вычислении критерия Ньютона.
# Поток якобиана первого порядка, LF или Roe.
jacobian_flux = LF
# on — фиксировать максимальную скорость LF или матрицу диссипации Роу.
# Для Роу фиксируются собственные значения и векторы среднего состояния.
# Так для LF применяется идея фиксированной диссипативной матрицы из препринта
# П. А. Бахвалова 2020 г., № 79, с. 34–35 (https://keldysh.ru/papers/2020/prep2020_79_eng.pdf).
# off — дифференцировать полный поток, включая изменение диссипации.
simplified_jacobian = off
# BJ — блочный Якоби. CSGS — прямой и обратный проходы блочного Гаусса–Зейделя.
# Процессы MPI обмениваются результатом прямого прохода перед обратным.
preconditioner = BJ
# Относительная точность линейной системы: ||b - A*delta||_inf / ||b||_inf.
solver_tol = 1e-8
# Относительная точность нелинейного шага: ||F(U)||_inf / ||F(U_n)||_inf.
# Норму берём по всем узлам и компонентам. При нулевой исходной невязке шаг уже решён.
newton_tol = 1e-6
# Максимальное число поправок Ньютона за один шаг по времени.
newton_max_iter = 20
# Частота обновления пространственного якобиана. 0 — каждая поправка Ньютона.
# 1 — один раз за шаг, N — один раз за N принятых шагов, если потребовалась поправка.
# При изменении dt временной вклад в диагональ пересчитывается на текущем шаге.
jacobian_update_every = 0

# Установление стационарного решения (main.txt: time_scheme = steady).
# В каждом узле делим сумму исходящих потоков на объём. M — максимум модуля
# полученной величины по всем уравнениям и собственным узлам всех процессов.
# relative: M/M_начальное <= steady_tol; absolute: M <= steady_tol.
# После перезапуска начальную норму берём от поля из контрольной точки.
steady_criterion = relative
steady_tol = 1e-6
# Число псевдошагов за запуск. При исчерпании записываются поля и возвращается ошибка.
steady_max_steps = 10000
