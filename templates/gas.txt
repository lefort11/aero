# gamma обязателен для Euler и Linear. Значения по умолчанию нет.

# gamma — показатель адиабаты идеального газа. Обязательное конечное число > 1.
gamma = 1.4

# mach — число Маха набегающего потока. Обязательно для initial_type = freestream и для вязкого расчёта.
# Значения по умолчанию нет. При rho_inf = 1 и |u_inf| = 1 скорость звука равна 1/mach.
# mach = 0.5
# Для автоматического reference_state на границе нужны gamma и mach > 0.
# Необязательные углы в градусах: phi — азимут в xy, theta — возвышение от xy.
# Оба по умолчанию равны нулю, тогда поток направлен вдоль +x. Углы начального поля задаются в init.txt.
# phi = 0
# theta = 0

# При viscosity = normal или sutherland обязательны reynolds, prandtl и t_amb.
# Значений по умолчанию для них нет. Ниже приведён пример безразмерной постановки.
# reynolds — число Рейнольдса Re = rho_ref*U_ref*L_ref/mu(T_amb) > 0.
# Используйте rho_ref = rho_inf, U_ref = |u_inf| и ту же L_ref, которой обезразмерены расчётные координаты.
# reynolds = 1e6
# prandtl — молекулярное число Прандтля Pr > 0, отношение кинематической вязкости к температуропроводности.
# prandtl = 0.72
# t_amb — абсолютная опорная температура газа в кельвинах, t_amb > 0.
# t_amb = 273.15

# Температурный закон задаёт отношение вязкостей f = mu(T)/mu(T_amb).
# Отношение температур вычисляется из местного состояния: theta = gamma*mach^2*p/rho.
# При theta = T/t_amb оба закона дают f = 1 для theta = 1.
# normal:     f = sqrt(theta).
# sutherland: f = theta^(3/2) * (1+s)/(theta+s), где s = sutherland_s/t_amb.
# sutherland_s — температурная постоянная Сазерленда, К. Обязательна только для sutherland.
# Значения по умолчанию нет. При sutherland_s = 0 закон сводится к normal.
# sutherland_s = 110.4

# Все заданные параметры должны быть конечными, sutherland_s >= 0.
# Для невязкого расчёта достаточно gamma. Режим initial_type = freestream дополнительно требует mach.
